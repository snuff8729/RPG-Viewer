# RPG Viewer — 이 맵: 실제 맵 그림 + 가는 길 + 구분되는 색

**User feedback (verbatim):** "이 맵보는거 너무 헷갈려(맵 생긴거랑 달라서도 있고, 벽이랑 아닌거 구분도 안되고 특정 픽셀 맞춰서 서야하는데 그게쉽지 않음), 지금 할 수 있음, 반복하면 달라짐, 이거 색도 넘 구분 안되고."

**Decision:** replace the dot grid with the real map drawn from the game's tilesets (incl. autotiles) and event character sprites, overlay numbered markers, and when an event is selected draw the walking route from the player plus the exact tile to stand on and the direction to face. Use clearly different colours + icons per section, consistent between map markers, cards and sidebar.

## Contract (core → UI)
```ts
// src/core/map/tiles.ts
export interface TilesetInfo { id: number; mode: number; names: string[] /* 9: A1..A5,B,C,D,E */; flags: number[] }
export interface MapTiles { mapId: number; width: number; height: number; tilesetId: number; data: number[] /* width*height*6 */ }
export function loadMapTiles(folder: GameFolder, data: GameData, mapId: number): Promise<{ map: MapTiles; tileset: TilesetInfo } | null>; // reads Map###.json + Tilesets.json on demand (cache Tilesets per data); never throws
export interface DrawOp { sheet: number /* 0..8 index into tileset.names */; sx: number; sy: number; dx: number; dy: number; w: number; h: number }
export function tileSize(engine: "MV" | "MZ"): number; // 48 for both
export function drawOpsForTile(tileId: number, x: number, y: number, size: number, tileset: TilesetInfo, mapData?: MapTiles): DrawOp[]; // port of rpg_core Tilemap._drawTile/_drawAutotile/_drawNormalTile (FLOOR/WALL/WATERFALL autotile tables, A1 animation frame 0, A3/A4 wall rules, shadows from layer 4 drawn as half-tile dark rects are OPTIONAL: return them as sheet -1 ops)
export function layerOps(map: MapTiles, tileset: TilesetInfo, size: number): { lower: DrawOp[]; upper: DrawOp[] }; // upper = tiles with flag 0x10 (star)
// src/core/map/passage.ts
export function passable(map: MapTiles, tileset: TilesetInfo, x: number, y: number, dir: 2|4|6|8): boolean; // RPG Maker checkPassage rules over layers 3..0 (flag bits 0x01 down,0x02 left,0x04 right,0x08 up; 0x10 star skip)
export function blocked(map: MapTiles, tileset: TilesetInfo, x: number, y: number): boolean; // impassable in all 4 dirs (for wall shading)
// src/core/map/route.ts
export interface EventStand { x: number; y: number; face: 2|4|6|8 | null /* null = step on the event tile */ }
export function standTiles(ev: { x: number; y: number; priorityType: number; trigger: number; through: boolean }, map: MapTiles, tileset: TilesetInfo): EventStand[]; // priority 1 (same as characters) & trigger 0: the 4 neighbours facing the event (also across a counter tile flag 0x80: 2 tiles away); priority 0/2 or touch triggers: the event tile itself
export interface Route { steps: { x: number; y: number }[]; stand: EventStand; text: string }
export function routeTo(map: MapTiles, tileset: TilesetInfo, from: { x: number; y: number }, stands: EventStand[], blockers: Set<string> /* "x,y" of other solid events */): Route | null; // BFS 4-dir with passable(); text e.g. "오른쪽 3칸, 위 2칸 가서 위를 보고 확인" (summarised as net dx/dy + facing; for stand-on: "…가서 그 칸에 서기")
```
Event page graphics come from `MapEvent.pages[p].image` `{ tileId, characterName, characterIndex, direction, pattern }` and `priorityType`, `through`, `trigger`. Keep these fields in the loader (they're already in map JSON events; verify the types expose them, add to `EventPage` type if missing).
